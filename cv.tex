    \begin{project}{Prediction of Thermodynamic Properties Based on Functional Groups}{May 2025 -- Jul.\ 2025}{}
        \item Developed and benchmarked classical machine learning algorithms and neural networks to predict thermodynamic properties of molecules from quantum features. \href{https://github.com/SinaGhanbarii/Prediction-of-thermodynamic-properties-based-on-functional-groups}{\textbf{[GitHub]}}
    \end{project}

    \begin{project}{Techno-Economic Analysis and LCA of Biochar Production}{Apr.\ 2025 -- May 2025}{}
        \item Designed a pilot-scale slow-pyrolysis plant converting corncob waste into biochar, using syngas for internal heating and sizing key equipment (rotary kiln, dryers)
        \item Performed a Class~5 techno-economic analysis (CAPEX, OPEX, NPV, IRR, ROI) with sensitivity analysis on feed rate and product pricing
        \item Conducted a cradle-to-gate Life Cycle Assessment (LCA) quantifying GHG emissions (CO\textsubscript{2}, CH\textsubscript{4}) under different electricity-mix scenarios
    \end{project}

    \begin{project}{Conceptual Design and Economic Feasibility of HDA of Toluene Plant}{Mar.\ 2024 -- Apr.\ 2024}{}
        \item Designed and simulated a hydrodealkylation (HDA) plant in MATLAB and Aspen HYSYS, and assessed its economic feasibility using NPV, IRR, ROI, and PBP. \href{https://github.com/SinaGhanbarii/HDA-Plant-Simulation}{\textbf{[GitHub]}}
    \end{project}

    \begin{project}{Numerical Methods and Their Application in Chemical Engineering}{Aug.\ 2023 -- Present}{}
        \item Created and maintain a GitHub repository of educational material, code samples, and chemical engineering applications of numerical methods. \href{https://github.com/SinaGhanbarii/Numerical-Methods-and-Their-Application-in-Chemical-Engineering}{\textbf{[GitHub]}}
    \end{project}

    \begin{project}{Technical and Financial Analysis of 2-Ethylhexyl Acrylate Plant}{Jan.\ 2023 -- Feb.\ 2023}{}
        \item Designed and simulated a 2-ethylhexyl acrylate plant in Aspen HYSYS, covering the reaction, separation, and heat-integration systems
        \item Carried out a market study and calculated NPV, IRR, ROI, and PBP over the plant lifetime
